cmake_minimum_required(VERSION 3.16)
project(webos_multiview_probe VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(probe_inventory native/probe/inventory.cpp)
target_include_directories(probe_inventory PUBLIC native/probe)
target_compile_features(probe_inventory PUBLIC cxx_std_11)
target_compile_options(probe_inventory PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(multiview-probe native/probe/main.cpp)
target_link_libraries(multiview-probe PRIVATE probe_inventory)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(inventory-test tests/host/inventory_test.cpp)
  target_link_libraries(inventory-test PRIVATE probe_inventory)
  add_test(NAME inventory COMMAND inventory-test)
endif()
install(TARGETS multiview-probe RUNTIME DESTINATION bin)
