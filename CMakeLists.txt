cmake_minimum_required(VERSION 3.16)
project(webos_multiview_probe VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(probe_inventory native/probe/inventory.cpp)
target_include_directories(probe_inventory PUBLIC native/probe)
target_compile_features(probe_inventory PUBLIC cxx_std_11)
target_compile_options(probe_inventory PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(multiview-probe native/probe/main.cpp)
target_link_libraries(multiview-probe PRIVATE probe_inventory)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(hdmi1-ownership-test tests/host/hdmi1_ownership_test.cpp
      native/phase2-media-registration/ownership.cpp)
  target_include_directories(hdmi1-ownership-test PRIVATE tests/host/ownership-sdk)
  target_compile_features(hdmi1-ownership-test PRIVATE cxx_std_14)
  target_compile_options(hdmi1-ownership-test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  add_test(NAME hdmi1-ownership-lifetime COMMAND hdmi1-ownership-test)
  set_tests_properties(hdmi1-ownership-lifetime PROPERTIES TIMEOUT 5)
  add_executable(inventory-test tests/host/inventory_test.cpp)
  target_link_libraries(inventory-test PRIVATE probe_inventory)
  add_test(NAME inventory COMMAND inventory-test)
  foreach(check IN ITEMS progress_watch veto_error)
    add_executable(${check}-test tests/host/${check}_test.cpp)
    target_compile_features(${check}-test PRIVATE cxx_std_14)
    # These existing acceptance tests use assert; keep it active in Release too.
    target_compile_options(${check}-test PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    add_test(NAME ${check} COMMAND ${check}-test)
  endforeach()
endif()
install(TARGETS multiview-probe RUNTIME DESTINATION bin)
