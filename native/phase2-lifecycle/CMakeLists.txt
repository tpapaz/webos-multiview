cmake_minimum_required(VERSION 3.16)
project(multiview_lifecycle LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(multiview-lifecycle main.c)
target_compile_options(multiview-lifecycle PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(multiview-lifecycle PROPERTIES SKIP_BUILD_RPATH TRUE)
if(CMAKE_CROSSCOMPILING AND NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use the existing 32-bit ARM webOS SDK")
endif()
