cmake_minimum_required(VERSION 3.16)
project(multiview_admission LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(CMAKE_CROSSCOMPILING)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use the 32-bit ARM webOS SDK")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  find_library(LUNA_LIBRARY luna-service2 REQUIRED)
  find_library(JSON_C_LIBRARY json-c REQUIRED)
  add_executable(multiview-admission main.cpp admission.cpp)
  target_link_libraries(multiview-admission PRIVATE PkgConfig::GLIB "${LUNA_LIBRARY}" "${JSON_C_LIBRARY}")
  # Same retail C1 shared-library allowance as Phase 1; requires offline ABI check.
  target_link_options(multiview-admission PRIVATE "-Wl,--allow-shlib-undefined")
  set_target_properties(multiview-admission PROPERTIES SKIP_BUILD_RPATH TRUE)
elseif(BUILD_TESTING)
  find_path(JSON_C_INCLUDE_DIR json-c/json.h)
  if(NOT JSON_C_INCLUDE_DIR AND SDK_JSON_HEADERS)
    # Copy just the architecture-independent json-c API headers, never the ARM libc headers.
    file(COPY "${SDK_JSON_HEADERS}/json-c" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include")
    set(JSON_C_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/include")
  endif()
  find_library(JSON_C_LIBRARY NAMES json-c libjson-c.so.5 REQUIRED)
  if(NOT JSON_C_INCLUDE_DIR)
    message(FATAL_ERROR "Install host json-c headers or supply SDK_JSON_HEADERS")
  endif()
  add_executable(admission-test ../../tests/host/admission_test.cpp admission.cpp)
  target_include_directories(admission-test PRIVATE "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(admission-test PRIVATE "${JSON_C_LIBRARY}")
  target_compile_options(admission-test PRIVATE -UNDEBUG)
  add_test(NAME admission COMMAND admission-test)
  add_executable(admission-report-test ../../tests/host/admission_report_test.cpp)
  target_compile_options(admission-report-test PRIVATE -UNDEBUG)
  add_test(NAME admission-report-safety COMMAND admission-report-test)
endif()
