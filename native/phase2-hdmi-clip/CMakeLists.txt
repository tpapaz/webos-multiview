cmake_minimum_required(VERSION 3.16)
project(multiview_hdmi_clip LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(_GNU_SOURCE)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
find_path(JSON_C_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSON_C_LIBRARY NAMES json-c libjson-c.so.5 REQUIRED)
set(PROBE_SOURCES worker.cpp ../phase2-hdmi-picture/session.cpp)

if(CMAKE_CROSSCOMPILING)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use the existing ARM32 webOS SDK")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  find_library(LUNA_LIBRARY luna-service2 REQUIRED)
  find_library(CLOCK_LIBRARY rt REQUIRED)
  find_library(PTHREAD_LIBRARY pthread REQUIRED)
  find_library(SMP_LIBRARY playerAPIs REQUIRED)
  add_executable(multiview-hdmi-clip main.cpp ${PROBE_SOURCES})
  target_include_directories(multiview-hdmi-clip PRIVATE "${JSON_C_INCLUDE_DIR}")
  target_link_options(multiview-hdmi-clip PRIVATE -Wl,--no-as-needed -Wl,--allow-shlib-undefined)
  target_link_libraries(multiview-hdmi-clip PRIVATE PkgConfig::SDL2 PkgConfig::GLIB
    "${LUNA_LIBRARY}" "${JSON_C_LIBRARY}" "${CLOCK_LIBRARY}" "${PTHREAD_LIBRARY}" "${SMP_LIBRARY}")
  set_target_properties(multiview-hdmi-clip PROPERTIES SKIP_BUILD_RPATH TRUE)
elseif(BUILD_TESTING)
  add_executable(hdmi-clip-test ../../tests/host/hdmi_clip_test.cpp ${PROBE_SOURCES})
  target_include_directories(hdmi-clip-test PRIVATE "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(hdmi-clip-test PRIVATE "${JSON_C_LIBRARY}")
  target_compile_options(hdmi-clip-test PRIVATE -UNDEBUG)
  add_test(NAME hdmi-clip-bounded-playback COMMAND hdmi-clip-test)
  set_tests_properties(hdmi-clip-bounded-playback PROPERTIES TIMEOUT 20)
endif()
