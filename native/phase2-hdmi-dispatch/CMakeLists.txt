cmake_minimum_required(VERSION 3.16)
project(multiview_hdmi_adapter_acceptance LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(_GNU_SOURCE)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
find_path(JSON_C_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSON_C_LIBRARY NAMES json-c libjson-c.so.5 REQUIRED)
set(SESSION ../phase2-hdmi-picture/session.cpp)
if(CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Build the deployed app from phase2-hdmi-candidate-0.1.12; this directory owns host acceptance only")
elseif(BUILD_TESTING)
  find_package(Threads REQUIRED)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_library(HOST_GLIB NAMES libglib-2.0.so.0 REQUIRED)
  add_executable(hdmi-session-test ../../tests/host/hdmi_window_test.cpp ${SESSION})
  target_include_directories(hdmi-session-test PRIVATE "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(hdmi-session-test PRIVATE "${JSON_C_LIBRARY}")
  target_compile_options(hdmi-session-test PRIVATE -UNDEBUG)
  add_test(NAME hdmi-session-rules COMMAND hdmi-session-test)
  add_executable(callback-mailbox-test ../../tests/host/callback_mailbox_test.cpp)
  target_link_libraries(callback-mailbox-test PRIVATE Threads::Threads)
  target_compile_options(callback-mailbox-test PRIVATE -UNDEBUG)
  add_test(NAME callback-mailbox-concurrency COMMAND callback-mailbox-test)
  add_executable(physical-adapter-test ../../tests/host/physical_adapter_dispatch_test.cpp ${SESSION})
  target_include_directories(physical-adapter-test PRIVATE ../../tests/host/hdmi-dispatch-headers "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(physical-adapter-test PRIVATE "${HOST_GLIB}" "${JSON_C_LIBRARY}" Threads::Threads)
  target_compile_options(physical-adapter-test PRIVATE -UNDEBUG)
  add_test(NAME physical-adapter-callback-lifecycle COMMAND physical-adapter-test)
  add_test(NAME physical-adapter-retained-owner COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/host/physical_adapter_exit_test.py" "$<TARGET_FILE:physical-adapter-test>")
  add_executable(parent-witness-test ../../tests/host/parent_witness_test.cpp)
  target_compile_options(parent-witness-test PRIVATE -UNDEBUG)
  add_test(NAME parent-death-witness COMMAND parent-witness-test "${CMAKE_CURRENT_BINARY_DIR}")
  set_tests_properties(callback-mailbox-concurrency physical-adapter-callback-lifecycle physical-adapter-retained-owner parent-death-witness PROPERTIES TIMEOUT 15)
endif()
