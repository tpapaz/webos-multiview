cmake_minimum_required(VERSION 3.16)
project(multiview_physical_candidate_0_1_12 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(_GNU_SOURCE)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
find_path(JSON_C_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSON_C_LIBRARY NAMES json-c libjson-c.so.5 REQUIRED)
set(SESSION ../phase2-hdmi-picture/session.cpp)
if(CMAKE_CROSSCOMPILING)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use the existing ARM32 webOS SDK")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  find_library(LUNA_LIBRARY luna-service2 REQUIRED)
  find_library(CLOCK_LIBRARY rt REQUIRED)
  find_library(PTHREAD_LIBRARY pthread REQUIRED)
  add_executable(multiview-hdmi-window main.cpp ${SESSION})
  target_include_directories(multiview-hdmi-window PRIVATE "${JSON_C_INCLUDE_DIR}")
  target_link_options(multiview-hdmi-window PRIVATE -Wl,--no-as-needed -Wl,--allow-shlib-undefined)
  target_link_libraries(multiview-hdmi-window PRIVATE PkgConfig::SDL2 PkgConfig::GLIB "${LUNA_LIBRARY}" "${JSON_C_LIBRARY}" "${CLOCK_LIBRARY}" "${PTHREAD_LIBRARY}")
  set_target_properties(multiview-hdmi-window PROPERTIES SKIP_BUILD_RPATH TRUE)
elseif(BUILD_TESTING)
  find_library(HOST_GLIB NAMES libglib-2.0.so.0 REQUIRED)
  add_executable(physical-candidate-test ../../tests/host/physical_candidate_test.cpp ${SESSION})
  target_include_directories(physical-candidate-test PRIVATE ../../tests/host/physical-candidate-headers "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(physical-candidate-test PRIVATE "${JSON_C_LIBRARY}" "${HOST_GLIB}")
  target_compile_options(physical-candidate-test PRIVATE -UNDEBUG)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/trace-fixtures")
  add_test(NAME physical-candidate-contract COMMAND physical-candidate-test "${CMAKE_CURRENT_BINARY_DIR}/trace-fixtures")
  add_executable(physical-candidate-witness-test ../../tests/host/physical_candidate_witness_test.cpp ${SESSION})
  target_include_directories(physical-candidate-witness-test PRIVATE ../../tests/host/physical-candidate-headers "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(physical-candidate-witness-test PRIVATE "${JSON_C_LIBRARY}" "${HOST_GLIB}")
  target_compile_options(physical-candidate-witness-test PRIVATE -UNDEBUG)
  add_test(NAME physical-candidate-bound-witness COMMAND physical-candidate-witness-test "${CMAKE_CURRENT_BINARY_DIR}")
  set_tests_properties(physical-candidate-contract physical-candidate-bound-witness PROPERTIES TIMEOUT 8)
endif()
