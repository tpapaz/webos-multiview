cmake_minimum_required(VERSION 3.16)
project(multiview_hdmi_window LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
find_package(PkgConfig REQUIRED)
find_path(JSON_C_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSON_C_LIBRARY NAMES json-c libjson-c.so.5 REQUIRED)
if(CMAKE_CROSSCOMPILING)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use the 32-bit ARM webOS SDK")
  endif()
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  find_library(LUNA_LIBRARY luna-service2 REQUIRED)
  add_executable(multiview-hdmi-window main.cpp session.cpp)
  target_include_directories(multiview-hdmi-window PRIVATE "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(multiview-hdmi-window PRIVATE PkgConfig::SDL2 PkgConfig::GLIB "${LUNA_LIBRARY}" "${JSON_C_LIBRARY}")
  # Retail C1 link inputs only; no firmware libraries are bundled. Offline ABI
  # validation is required, as for the retained admission/endurance controls.
  target_link_options(multiview-hdmi-window PRIVATE "-Wl,--allow-shlib-undefined")
  set_target_properties(multiview-hdmi-window PROPERTIES SKIP_BUILD_RPATH TRUE)
elseif(BUILD_TESTING)
  add_executable(hdmi-window-test ../../tests/host/hdmi_window_test.cpp session.cpp)
  target_include_directories(hdmi-window-test PRIVATE "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(hdmi-window-test PRIVATE "${JSON_C_LIBRARY}")
  target_compile_options(hdmi-window-test PRIVATE -UNDEBUG)
  add_test(NAME hdmi-window-acceptance COMMAND hdmi-window-test)
endif()
