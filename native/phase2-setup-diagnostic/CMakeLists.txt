cmake_minimum_required(VERSION 3.16)
project(multiview_entry_diagnostic LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -ffunction-sections -fdata-sections)
add_link_options(-Wl,--gc-sections)
find_path(JSON_C_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSON_C_LIBRARY NAMES json-c libjson-c.so.5 REQUIRED)
set(PRODUCTION_SESSION ../phase2-hdmi-window/session.cpp)
if(CMAKE_CROSSCOMPILING)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use the existing 32-bit ARM webOS SDK")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  find_library(LUNA_LIBRARY luna-service2 REQUIRED)
  find_library(CLOCK_LIBRARY rt REQUIRED)
  add_executable(multiview-hdmi-window main.cpp ${PRODUCTION_SESSION})
  target_include_directories(multiview-hdmi-window PRIVATE "${JSON_C_INCLUDE_DIR}")
  # Retain the candidate's direct loader inputs, with no SDL/LS2 function references.
  target_link_options(multiview-hdmi-window PRIVATE -Wl,--no-as-needed -Wl,--allow-shlib-undefined)
  target_link_libraries(multiview-hdmi-window PRIVATE PkgConfig::SDL2 PkgConfig::GLIB "${LUNA_LIBRARY}" "${JSON_C_LIBRARY}" "${CLOCK_LIBRARY}")
  set_target_properties(multiview-hdmi-window PROPERTIES SKIP_BUILD_RPATH TRUE)
elseif(BUILD_TESTING)
  add_executable(entry-diagnostic-test ../../tests/host/setup_diagnostic_test.cpp ${PRODUCTION_SESSION})
  target_include_directories(entry-diagnostic-test PRIVATE "${JSON_C_INCLUDE_DIR}")
  target_link_libraries(entry-diagnostic-test PRIVATE "${JSON_C_LIBRARY}")
  target_compile_options(entry-diagnostic-test PRIVATE -UNDEBUG)
  add_test(NAME entry-diagnostic-acceptance COMMAND entry-diagnostic-test)
endif()
